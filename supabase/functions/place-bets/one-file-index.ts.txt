// ------------------------------------------------------------------ money math
/** Kalshi taker fee for a whole order: 7% × C × P × (1−P), rounded up to the cent. */
export function kalshiFee(p: number, count: number): number {
  return Math.ceil(0.07 * count * p * (1 - p) * 100 - 1e-9) / 100;
}

/** Kalshi maker fee on a resting order that fills later: 0.0175 × C × P × (1−P), rounded up to the cent.
 * The fee schedule (July 2026) lists maker fees only for some series; we assume it applies, to be safe. */
export function kalshiMakerFee(p: number, count: number): number {
  return Math.ceil(0.0175 * count * p * (1 - p) * 100 - 1e-9) / 100;
}

/** The most we'd rest an order at: the price where the bet still returns `minEv` after the maker fee,
 * rounded DOWN to a whole cent. win/push are the model's chances at this exact number. */
export function restingPrice(win: number, push: number, minEv: number): number {
  const allIn = win / (1 - push + minEv);              // most we can pay per contract, fee included
  let p = allIn;
  for (let i = 0; i < 5; i++) p = allIn - 0.0175 * p * (1 - p);
  return Math.max(0.01, Math.min(0.99, Math.floor((p - 1e-9) * 100) / 100));
}

/** Polymarket US taker fee: 0.0695 × C × p × (1−p), banker's rounding to the cent. */
export function polyFee(p: number, count: number): number {
  const cents = 6.95 * count * p * (1 - p);
  const fl = Math.floor(cents), diff = cents - fl;
  const r = Math.abs(diff - 0.5) < 1e-9 ? (fl % 2 === 0 ? fl : fl + 1) : Math.round(cents);
  return r / 100;
}

/** Fewest contracts that win at least `toWin` dollars after the fee, buying at `price`. */
export function contractsToWin(price: number, toWin: number, fee: (p: number, c: number) => number): number {
  let c = Math.max(1, Math.ceil(toWin / (1 - price) - 1e-9));
  while (c * (1 - price) - fee(price, c) < toWin - 1e-9 && c < 100000) c++;
  return c;
}

/** Expected return per $1 at an all-in cost per contract, from the model's win and break-even cost. */
export function evAt(q: Quote, costPerContract: number): number {
  const push = Math.max(0, 1 - q.win / q.max_cost);
  return q.win * (1 / costPerContract - 1) - (1 - q.win - push);
}

// Normal distribution helpers (results land around the line with the measured SD).
export function normCdf(x: number): number {
  const t = 1 / (1 + 0.2316419 * Math.abs(x));
  const d = 0.3989422804014327 * Math.exp(-x * x / 2);
  const p = d * t * (0.31938153 + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429))));
  return x >= 0 ? 1 - p : p;
}
export function normInv(p: number): number {     // Acklam
  const a = [-39.69683028665376, 220.9460984245205, -275.9285104469687, 138.357751867269, -30.66479806614716, 2.506628277459239];
  const b = [-54.47609879822406, 161.5858368580409, -155.6989798598866, 66.80131188771972, -13.28068155288572];
  const c = [-0.007784894002430293, -0.3223964580411365, -2.400758277161838, -2.549732539343734, 4.374664141464968, 2.938163982698783];
  const d = [0.007784695709041462, 0.3224671290700398, 2.445134137142996, 3.754408661907416];
  const lo = 0.02425, hi = 1 - lo;
  if (p < lo) { const q = Math.sqrt(-2 * Math.log(p)); return (((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) / ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1); }
  if (p > hi) { const q = Math.sqrt(-2 * Math.log(1 - p)); return -(((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) / ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1); }
  const q = p - 0.5, r = q * q;
  return (((((a[0]*r+a[1])*r+a[2])*r+a[3])*r+a[4])*r+a[5])*q / (((((b[0]*r+b[1])*r+b[2])*r+b[3])*r+b[4])*r+1);
}

// What each extra point of spread is worth, MEASURED on six seasons (6,979 games, closing lines) — the
// same table as engine/pricing.py POINT_GAIN: cover-chance gain for d = 0.5 ... 7 extra points.
export const POINT_GAIN: Record<string, number[]> = {
  "close within 3": [0.014, 0.028, 0.040, 0.054, 0.071, 0.085, 0.097, 0.109, 0.122, 0.137, 0.154, 0.167, 0.181, 0.196],
  "3.5 to 7": [0.021, 0.039, 0.056, 0.074, 0.093, 0.108, 0.122, 0.134, 0.147, 0.160, 0.174, 0.187, 0.198, 0.208],
  "7.5+": [0.021, 0.039, 0.056, 0.075, 0.092, 0.110, 0.127, 0.145, 0.162, 0.178, 0.193, 0.208, 0.221, 0.233],
};
export function pointGain(absLine: number, d: number): number {
  const g = POINT_GAIN[absLine <= 3 ? "close within 3" : absLine <= 7 ? "3.5 to 7" : "7.5+"];
  const a = Math.abs(d);
  let val: number;
  if (a <= 7) {
    const i = a * 2, lo = Math.floor(i);
    const glo = lo === 0 ? 0 : g[lo - 1], ghi = g[Math.min(lo, 13)];
    val = glo + (ghi - glo) * (i - lo);
  } else val = g[13] + (g[13] - g[11]) * (a - 7);
  return d >= 0 ? val : -val;
}

/** (win, push) at `line`, given the model's win rate at its base line — same math as the Python card. */
export function outcomeAt(m: Model, line: number): { win: number; push: number } {
  const z0 = normInv(Math.min(Math.max(m.win0, 0.01), 0.99));
  const over = m.bet === "total" && m.side === "OVER";
  const cover = (L: number) => m.bet !== "total"
    ? Math.min(Math.max(m.win0 + pointGain(Math.abs(m.base), L - m.base), 0.01), 0.99)
    : normCdf(z0 + ((over ? m.base - L : L - m.base) / m.sd));
  if (Number.isInteger(line)) {
    const [worse, better] = over ? [line + 0.5, line - 0.5] : [line - 0.5, line + 0.5];
    const w = cover(worse);
    return { win: w, push: cover(better) - w };
  }
  return { win: cover(line), push: 0 };
}

export const americanCost = (a: number) => (a > 0 ? 100 / (a + 100) : -a / (-a + 100));

/** A clear refusal from the site (HTTP 4xx: not enough money, bad price, market closed...). Nothing was placed.
 * Timeouts, network errors and 5xx stay "unknown" and are never retried. */
export function rejected(e: unknown): boolean {
  return /→ 4\d\d\b/.test(msg(e)) && !/→ 40[89]\b|→ 425\b|→ 429\b/.test(msg(e));
}

// Caden (2026-10-22): when two prices make about the same money, take the one that wins more often.
export const POINTS_TIEBREAK = 0.02;
/** Best first; among quotes within POINTS_TIEBREAK of the top expected return, the most likely to cash goes first. */
export function preferPoints(qs: Quote[]): Quote[] {
  const out = [...qs].sort((a, b) => b.ev - a.ev);
  if (!out.length) return out;
  const top = out[0].ev;
  const cash = (q: Quote) => q.win + Math.max(0, 1 - q.win / q.max_cost) / 2;
  const near = out.filter((q) => q.ev >= top - POINTS_TIEBREAK).sort((a, b) => cash(b) - cash(a) || b.ev - a.ev);
  return near.concat(out.filter((q) => q.ev < top - POINTS_TIEBREAK));
}

/** The price the model saw for this quote (what the limit order is capped at). */
export function modelPrice(q: Quote): number {
  return q.venue === "kalshi" ? Number(q.price) : Number(q.cost);
}

export function etDayStart(now: Date): Date {
  const parts = new Intl.DateTimeFormat("en-US", {
    timeZone: "America/New_York", year: "numeric", month: "2-digit", day: "2-digit",
    hour: "2-digit", minute: "2-digit", second: "2-digit", hourCycle: "h23",
  }).formatToParts(now);
  const get = (t: string) => Number(parts.find((p) => p.type === t)!.value);
  const etNowAsUtc = Date.UTC(get("year"), get("month") - 1, get("day"), get("hour"), get("minute"), get("second"));
  const offset = etNowAsUtc - Math.floor(now.getTime() / 1000) * 1000;   // ET minus UTC
  return new Date(Date.UTC(get("year"), get("month") - 1, get("day")) - offset);
}

// ------------------------------------------------------------------ the run
export type RunResult = { mode: string; actions: string[]; skips: { pick: string; reason: string }[]; stopped?: string };

export async function runOnce(db: Db, venues: Record<string, Venue>, now = new Date(),
                              keyCheck?: () => Promise<string>): Promise<RunResult> {
  const s = await db.settings();
  const live = !!s.live_betting_on;
  const mode = live ? "live" : "dry_run";
  const res: RunResult = { mode, actions: [], skips: [] };
  const on = (s.venues || []).filter((v) => venues[v]);

  // Rule 3: daily loss cap (only meaningful once real orders exist)
  const dayStart = etDayStart(now);
  const recent = await db.orders(new Date(now.getTime() - 4 * 86400e3).toISOString());
  if (live) {
    let pnl = 0;
    for (const v of on) {
      const filled = recent.filter((o) => o.mode === "live" && o.venue === v && (o.fill_count ?? 0) > 0);
      const mine = new Set(filled.map((o) => o.market));
      if (!mine.size) continue;
      try {
        pnl += venues[v].gradeFromGames
          ? await pnlFromGames(db, filled, dayStart.getTime())
          : await venues[v].pnlToday(mine, dayStart.getTime());
      } catch (e) {
        res.stopped = `could not check today's P&L on ${v}: ${msg(e)}`;
        await db.log("error", `Stopped this run: ${res.stopped}`);
        return res;
      }
    }
    if (pnl <= -Math.abs(s.daily_loss_cap)) {
      const why = `Daily loss cap hit: today's P&L ${money(pnl)} vs cap ${money(-s.daily_loss_cap)}. Live betting switched OFF.`;
      await db.updateSettings({ live_betting_on: false, disabled_reason: why });
      await db.log("warn", why, { pnl });
      res.stopped = why;
      return res;
    }
  }

  const minUpdated = new Date(now.getTime() - s.max_feed_age_minutes * 60e3).toISOString();
  const minTip = new Date(now.getTime() + 2 * 60e3).toISOString();
  const allFeed = await db.feed(minUpdated, minTip);
  const feed = allFeed.filter((r) => r.status === "bet");
  const limits = allFeed.filter((r) => r.status === "limit");

  // what's already committed (live), or — in a dry run — what the dry run has "placed" today
  const counted = recent.filter((o) =>
    live ? o.mode === "live" && ["pending", "filled", "partial", "resting"].includes(o.status)
         : o.mode === "dry_run" && (o.created_at ?? "") >= dayStart.toISOString());
  const openNow = () => counted.filter((o) => (o.tip_utc ?? "") > new Date(now.getTime() - 4 * 3600e3).toISOString());
  const gameDollars = (g: string) => counted.filter((o) => o.game_id === g)
    .reduce((t, o) => t + (o.fill_cost ?? o.stake ?? 0), 0);

  // money committed on each site during THIS run: a site's reported balance can lag right after a fill
  const spent: Record<string, number> = {};
  const available = async (vn: string) => (await venues[vn].balance()) - (spent[vn] ?? 0);
  if (live) await manageResting(db, venues, recent, allFeed, now, res);
  await placeResting(db, venues, s, live, on, limits, recent, counted, openNow, gameDollars, now, res, available, spent);

  for (const row of feed.sort((a, b) => best(b) - best(a))) {
    const skip = (reason: string) => res.skips.push({ pick: `${row.pick} (${row.system}) ${row.away}@${row.home}`, reason });

    // Rule 5: never the same bet twice (live). A bet that errored is never retried automatically.
    const liveForBet = recent.filter((o) => o.mode === "live" && o.bet_key === row.key && LIVE_STATUSES.includes(o.status));
    if (live && liveForBet.length) { skip(`already ${liveForBet[0].status} (${liveForBet[0].venue})`); continue; }

    let quotes = (row.quotes || []).filter((q) => on.includes(q.venue) && !venues[q.venue].offers &&
      (q.venue !== "kalshi" || (q.ticker && q.order_side)) && (q.venue !== "polymarket_us" || q.ml));
    for (const vn of on) {                     // peer-to-peer books: price every live offer with the model
      if (!venues[vn].offers || !row.model) continue;
      try {
        quotes = quotes.concat(await venues[vn].offers!(row));
      } catch (e) {
        res.skips.push({ pick: `${row.pick} (${row.system}) ${row.away}@${row.home}`, reason: `${venues[vn].name} offers failed: ${msg(e)}` });
      }
    }
    quotes = preferPoints(quotes.filter((q) => q.ev >= s.min_ev));
    if (!quotes.length) {
      const notes = on.map((vn) => venues[vn].explain?.(row) ?? "").filter(Boolean).join(" · ");
      skip(`no quote on ${on.join("/") || "any switched-on venue"} with expected return ≥ ${pct(s.min_ev)}${notes ? ` — ${notes}` : ""}`);
      continue;
    }

    let done = false;
    const why: string[] = [];
    for (const q of quotes) {
      const v = venues[q.venue];
      const seen = modelPrice(q);
      let chk;
      try {
        chk = await v.ask(q, row);
      } catch (e) {
        why.push(`${v.name}: price check failed (${msg(e)})`);
        continue;
      }
      const market = chk.market;
      // Rule 5: never the same market twice
      const marketTaken = recent.find((o) => o.mode === "live" && o.venue === q.venue && o.market === market &&
        LIVE_STATUSES.includes(o.status));
      if (live && marketTaken) { why.push(`${v.name} ${market}: already ${marketTaken.status}`); continue; }
      const unfilled = recent.filter((o) => o.mode === "live" && o.venue === q.venue && o.market === market && o.status === "unfilled");
      if (live && unfilled.length >= MAX_UNFILLED_TRIES) { why.push(`${v.name} ${market}: ${unfilled.length} unfilled tries already`); continue; }

      // a dry run that already logged this bet just refreshes it (no duplicate "would have placed")
      const prior = live ? undefined
        : recent.find((o) => o.mode === "dry_run" && o.venue === q.venue && o.market === market && o.bet_key === row.key);
      if (prior?.id && chk.ask != null) {
        await db.updateOrder(prior.id, { ask_at_check: chk.ask, seen_count: (prior.seen_count ?? 1) + 1 });
        done = true;
        break;
      }

      // Rule 6: re-check the ask; never pay more than the model saw
      const ask = chk.ask;
      if (ask == null || ask <= 0 || ask >= 0.99) { why.push(`${v.name} ${q.contract}: nothing to buy right now`); continue; }
      if (ask > seen + 1e-9) { why.push(`${v.name} ${q.contract}: ask ${cents(ask)} moved above the model's ${cents(seen)}`); continue; }

      // Rule 2: same size every bet — to win a fixed amount (default), or a fixed stake
      const limit = round4(seen);
      const toWin = s.to_win_dollars && s.to_win_dollars > 0
        ? (row.to_win && row.to_win > 0 ? Math.min(row.to_win, s.to_win_dollars) : s.to_win_dollars) : 0;
      const count = toWin > 0
        ? contractsToWin(limit, toWin, (p, c) => v.fee(p, c))
        : Math.floor(s.stake_dollars / limit + 1e-9);
      if (count < 1) { why.push(`stake ${money(s.stake_dollars)} buys less than one contract at ${cents(limit)}`); continue; }
      const stake = round2(count * limit);
      const fee = v.fee(limit, count);
      const evNow = evAt(q, (stake + fee) / count);
      if (evNow < s.min_ev) { why.push(`${v.name} ${q.contract}: only ${pct(evNow)} after fees`); continue; }

      // Rule 4: open bets and dollars per game
      if (openNow().length >= s.max_open_bets) { why.push(`max open bets (${s.max_open_bets}) reached`); break; }
      if (gameDollars(row.game_id) + stake > s.max_dollars_per_game + 1e-9) {
        why.push(`would put ${money(gameDollars(row.game_id) + stake)} on this game (max ${money(s.max_dollars_per_game)})`); break;
      }

      const order: OrderRow = {
        mode, env: q.venue === "kalshi" ? s.kalshi_env : "prod", venue: q.venue, market, order_side: orderSide(q),
        bet_key: row.key, game_id: row.game_id, tip_utc: row.tip_utc, pick: row.pick, system: row.system,
        contract: q.contract, line: q.line, model_price: seen, ask_at_check: ask, limit_price: limit, count,
        stake, est_fee: fee, ev: round4(evNow), status: live ? "pending" : "dry_run", bet: row.bet, side: q.side,
      };
      const text = q.venue === "betopenly"
        ? `BetOpenly ACCEPT offer ${q.offer_id}: ${q.contract} at ${q.american! > 0 ? "+" : ""}${q.american} for ${money(stake)} ` +
          `to win ${money(round2(count - stake))} (fee ~${money(fee)}) — ${row.pick}, ${row.system}, ${row.away} @ ${row.home}, ` +
          `expected return ${pct(evNow)}` +
          (q.size != null && q.size < stake ? ` (offer only has ${money(q.size)} left — would take that)` : "")
        : `${q.venue_name ?? v.name} BUY ${order.order_side.toUpperCase()} ×${count} ${market} at ${cents(limit)} ` +
        `(stake ${money(stake)} + fee ${money(fee)}) — ${row.pick} via ${q.contract}, ${row.system}, ` +
        `${row.away} @ ${row.home}, expected return ${pct(evNow)}`;

      if (!live) {
        const ins = await db.insertOrder({ ...order, reason: "dry run: nothing sent" });
        await db.log("info", `WOULD HAVE PLACED: ${text}`, { ...order, id: ins.id });
        res.actions.push(`would have placed: ${text}`);
        counted.push({ ...order, created_at: now.toISOString() });
        done = true;
        break;
      }

      // live: Polymarket US stays dry until its NO-side price is verified
      if (q.venue === "polymarket_us" && !POLY_LIVE_VERIFIED) {
        why.push("Polymarket US is dry-run only until a preview order confirms how NO prices work");
        continue;
      }
      if (!v.hasKeys) { why.push(`${v.name}: no API keys in Supabase secrets`); continue; }
      try {
        const bal = await available(q.venue);
        if (bal < stake + fee) { why.push(`${v.name} balance ${money(bal)} is under ${money(stake + fee)}`); continue; }
      } catch (e) {
        why.push(`${v.name}: balance check failed (${msg(e)})`);
        continue;
      }
      // claim the market first — the unique index makes a second run's claim fail, so no double bet
      const ins = await db.insertOrder(order);
      if (ins.conflict || !ins.id) { why.push(`${v.name} ${market}: another run already claimed it`); continue; }
      try {
        const r = await v.order(order, ins.id, q);
        const status = r.fill_count >= count ? "filled" : r.fill_count > 0 ? "partial" : "unfilled";
        await db.updateOrder(ins.id, { status, order_id: r.order_id, fill_count: r.fill_count, fill_cost: r.fill_cost,
          fees: r.fees, response: r.raw, reason: status === "unfilled" ? "immediate-or-cancel: nothing filled at the limit" : null });
        await db.updateFeed(row.key, { order_status: status, order_venue: q.venue, order_id: r.order_id,
          order_fill_count: r.fill_count, order_fill_cost: r.fill_cost, order_placed_at: now.toISOString() });
        await db.log(status === "unfilled" ? "warn" : "info", `${status.toUpperCase()}: ${text}`,
          { order_id: r.order_id, fill_count: r.fill_count, fill_cost: r.fill_cost, fees: r.fees });
        res.actions.push(`${status}: ${text}`);
        if (r.fill_count > 0) counted.push({ ...order, status, fill_cost: r.fill_cost });
        spent[q.venue] = (spent[q.venue] ?? 0) + (r.fill_cost ?? 0) + (r.fees ?? 0);
        done = status !== "unfilled";
      } catch (e) {
        if (rejected(e)) {
          // the site refused the order outright (e.g. not enough money there): nothing was placed,
          // so it's safe to try the next-best site in this same run
          await db.updateOrder(ins.id, { status: "unfilled", reason: `rejected by ${v.name}: ${msg(e)}` });
          await db.log("warn", `REJECTED by ${v.name} (nothing placed, trying the next site): ${text}: ${msg(e)}`);
          why.push(`${v.name} refused it (${msg(e).slice(0, 120)})`);
          continue;
        }
        // Rule 7: anything else (timeout, server error) might have gone through — log it, never retry blindly;
        // the row stays 'error' and blocks this bet
        await db.updateOrder(ins.id, { status: "error", reason: msg(e) });
        await db.updateFeed(row.key, { order_status: "error", order_venue: q.venue });
        await db.log("error", `ORDER FAILED (not retried): ${text}: ${msg(e)}`);
        res.actions.push(`error: ${text}: ${msg(e)}`);
        done = true;
      }
      break;
    }
    if (!done) skip(why.join("; ") || "no quote passed the checks");
  }

  // test-only sites: price every card bet there too and log what we WOULD have placed, never place
  const watch = (s.watch_venues || []).filter((v) => venues[v] && !on.includes(v));
  for (const row of watch.length ? feed : []) {
    for (const vn of watch) {
      const v = venues[vn];
      try {
        const qs = (v.offers ? (row.model ? await v.offers(row) : []) : (row.quotes || []).filter((q) => q.venue === vn))
          .filter((q) => q.ev >= s.min_ev);
        const q = preferPoints(qs)[0];
        if (!q) continue;
        const chk = await v.ask(q, row);
        if (chk.ask == null || chk.ask > modelPrice(q) + 1e-9) continue;
        const prior = recent.find((o) => o.mode === "dry_run" && o.venue === vn && o.market === chk.market && o.bet_key === row.key);
        if (prior?.id) { await db.updateOrder(prior.id, { ask_at_check: chk.ask, seen_count: (prior.seen_count ?? 1) + 1 }); continue; }
        const limit = round4(modelPrice(q));
        const toWin = s.to_win_dollars && s.to_win_dollars > 0
          ? (row.to_win && row.to_win > 0 ? Math.min(row.to_win, s.to_win_dollars) : s.to_win_dollars) : 0;
        const count = toWin > 0 ? contractsToWin(limit, toWin, (p, c) => v.fee(p, c)) : Math.floor(s.stake_dollars / limit + 1e-9);
        if (count < 1) continue;
        const stake = round2(count * limit), fee = v.fee(limit, count), evNow = evAt(q, (stake + fee) / count);
        const order: OrderRow = {
          mode: "dry_run", env: "prod", venue: vn, market: chk.market, order_side: orderSide(q), bet_key: row.key,
          game_id: row.game_id, tip_utc: row.tip_utc, pick: row.pick, system: row.system, contract: q.contract, line: q.line,
          model_price: limit, ask_at_check: chk.ask, limit_price: limit, count, stake, est_fee: fee, ev: round4(evNow),
          status: "dry_run", bet: row.bet, side: q.side,
        };
        const text = `${v.name} (test site) ${q.contract} at ${q.american != null ? (q.american > 0 ? "+" : "") + q.american : cents(limit)} ` +
          `for ${money(stake)} to win ${money(round2(count - stake - fee))} — ${row.pick}, ${row.system}, ${row.away} @ ${row.home}, ` +
          `expected return ${pct(evNow)}`;
        const ins = await db.insertOrder({ ...order, reason: "test site: nothing sent" });
        await db.log("info", `WOULD HAVE PLACED: ${text}`, { ...order, id: ins.id });
        res.actions.push(`would have placed (test site): ${text}`);
      } catch (e) {
        res.skips.push({ pick: `${row.pick} (${row.system}) ${row.away}@${row.home}`, reason: `${v.name} (test site): ${msg(e)}` });
      }
    }
  }

  const last = s.last_heartbeat ? Date.parse(s.last_heartbeat) : 0;
  if (res.actions.length || now.getTime() - last >= HEARTBEAT_MINUTES * 60e3) {
    if (keyCheck) {
      // read-only: proves the keys work without placing anything
      try { await db.log("info", await keyCheck()); } catch (e) { await db.log("error", `Key check failed: ${msg(e)}`); }
    }
    await db.log("info", `run (${mode}${live ? ", " + s.kalshi_env : ""}): ${feed.length} bets on the card, ` +
      `${res.actions.length} action(s), ${res.skips.length} skipped`, { skips: res.skips });
    await db.updateSettings({ last_heartbeat: now.toISOString() });
  }
  return res;
}


const REST_STOP_BEFORE_TIP_MIN = 5;   // resting orders expire this many minutes before tip

/** Check every resting order: record fills, and cancel any whose bet is no longer a LIMIT on the card
 * (it became a BET, moved to PASS, was benched, or the game is about to start). */
async function manageResting(db: Db, venues: Record<string, Venue>, recent: OrderRow[], feed: FeedRow[], now: Date,
                             res: RunResult) {
  const limitKeys = new Set(feed.filter((r) => r.status === "limit").map((r) => r.key));
  for (const o of recent.filter((x) => x.mode === "live" && x.status === "resting" && x.order_id)) {
    const v = venues[o.venue];
    if (!v?.getOrder || !v.cancel) continue;
    try {
      let st = await v.getOrder(o.order_id!);
      const late = Date.parse(o.tip_utc) - now.getTime() < REST_STOP_BEFORE_TIP_MIN * 60e3;
      if (st.status === "resting" && (late || !limitKeys.has(o.bet_key))) {
        await v.cancel(o.order_id!);
        st = await v.getOrder(o.order_id!);
        const why = late ? "game about to start" : "no longer a LIMIT on the card";
        await db.log("info", `CANCELLED resting order (${why}): ${o.contract} for ${o.pick}; filled ${st.fill_count} of ${o.count}`);
        res.actions.push(`cancelled resting ${o.pick}: ${why}`);
      }
      const status = st.status === "resting" ? "resting"
        : st.fill_count >= o.count ? "filled" : st.fill_count > 0 ? "partial" : "unfilled";
      if (status !== o.status || st.fill_count !== (o.fill_count ?? 0)) {
        await db.updateOrder(o.id!, { status, fill_count: st.fill_count, fill_cost: st.fill_cost, fees: st.fees,
          reason: status === "unfilled" ? "resting order expired or cancelled with nothing filled" : null });
        if (st.fill_count > (o.fill_count ?? 0)) {
          await db.log("info", `FILLED (resting): ${o.contract} — ${st.fill_count} of ${o.count} for ${money(st.fill_cost)} — ${o.pick}`);
          await db.updateFeed(o.bet_key, { order_status: status, order_venue: o.venue, order_id: o.order_id,
            order_fill_count: st.fill_count, order_fill_cost: st.fill_cost, order_placed_at: now.toISOString() });
        }
        Object.assign(o, { status, fill_count: st.fill_count, fill_cost: st.fill_cost });
      }
    } catch (e) {
      await db.log("error", `Resting order check failed for ${o.pick} (${o.order_id}): ${msg(e)}`);
    }
  }
}

/** LIMIT bets: rest a good-till-canceled order on Kalshi at the most we'd pay (still +min_ev after the maker fee),
 * so the bet fills if the price comes to us. One resting order per bet; it expires before tip. */
async function placeResting(db: Db, venues: Record<string, Venue>, s: Settings, live: boolean, on: string[],
                            limits: FeedRow[], recent: OrderRow[], counted: OrderRow[], openNow: () => OrderRow[],
                            gameDollars: (g: string) => number, now: Date, res: RunResult,
                            available: (vn: string) => Promise<number> = async () => Infinity,
                            spent: Record<string, number> = {}) {
  const k = venues.kalshi;
  if (!on.includes("kalshi") || !k?.rest) return;
  for (const row of limits) {
    const skip = (reason: string) => res.skips.push({ pick: `${row.pick} (${row.system}) ${row.away}@${row.home}`, reason });
    const mode = live ? "live" : "dry_run";
    const mine = recent.filter((o) => o.bet_key === row.key && o.mode === mode &&
      (live ? LIVE_STATUSES.includes(o.status) : o.status === "dry_run"));
    if (mine.length) { skip(live ? `already ${mine[0].status} (${mine[0].venue})` : "resting order already logged"); continue; }
    const qs = (row.quotes || []).filter((q) => q.venue === "kalshi" && q.ticker && q.order_side && q.win && q.max_cost);
    if (!qs.length) { skip("LIMIT: no Kalshi market to rest an order on"); continue; }
    // the strike where resting at our price is worth the most
    const pick = qs.map((q) => {
      const push = Math.max(0, 1 - q.win / q.max_cost);
      const limit = restingPrice(q.win, push, s.min_ev);
      return { q, limit, ev: evAt(q, limit + kalshiMakerFee(limit, 100) / 100) };
    }).sort((a, b) => b.ev - a.ev)[0];
    const { q, limit } = pick;
    const toWin = s.to_win_dollars && s.to_win_dollars > 0
      ? (row.to_win && row.to_win > 0 ? Math.min(row.to_win, s.to_win_dollars) : s.to_win_dollars) : 0;
    const count = toWin > 0 ? contractsToWin(limit, toWin, kalshiMakerFee) : Math.floor(s.stake_dollars / limit + 1e-9);
    if (count < 1) continue;
    const stake = round2(count * limit), fee = kalshiMakerFee(limit, count);
    if (openNow().length >= s.max_open_bets) { skip(`max open bets (${s.max_open_bets}) reached`); continue; }
    if (gameDollars(row.game_id) + stake > s.max_dollars_per_game + 1e-9) {
      skip(`resting order would put ${money(gameDollars(row.game_id) + stake)} on this game (max ${money(s.max_dollars_per_game)})`); continue;
    }
    const expires = Date.parse(row.tip_utc) / 1000 - REST_STOP_BEFORE_TIP_MIN * 60;
    if (expires * 1000 <= now.getTime()) continue;
    const order: OrderRow = {
      mode, env: s.kalshi_env, venue: "kalshi", market: String(q.ticker), order_side: String(q.order_side),
      bet_key: row.key, game_id: row.game_id, tip_utc: row.tip_utc, pick: row.pick, system: row.system,
      contract: q.contract, line: q.line, model_price: limit, ask_at_check: null as unknown as number, limit_price: limit,
      count, stake, est_fee: fee, ev: round4(evAt(q, (stake + fee) / count)), status: live ? "pending" : "dry_run",
      bet: row.bet, side: q.side,
    };
    const text = `Kalshi RESTING ${order.order_side.toUpperCase()} ×${count} ${q.ticker} at ${cents(limit)} until ` +
      `${REST_STOP_BEFORE_TIP_MIN} min before tip (stake ${money(stake)}, maker fee up to ${money(fee)}) — ${row.pick} via ` +
      `${q.contract}, ${row.system}, ${row.away} @ ${row.home}, expected return ${pct(order.ev)} if it fills`;
    if (!live) {
      const ins = await db.insertOrder({ ...order, reason: "dry run: resting order not sent" });
      await db.log("info", `WOULD HAVE PLACED: ${text}`, { ...order, id: ins.id });
      res.actions.push(`would have placed: ${text}`);
      continue;
    }
    try {
      const bal = await available("kalshi");
      if (bal < stake + fee) { skip(`resting order: Kalshi balance ${money(bal)} is under ${money(stake + fee)}`); continue; }
    } catch (e) { skip(`resting order: Kalshi balance check failed (${msg(e)})`); continue; }
    const ins = await db.insertOrder(order);
    if (ins.conflict || !ins.id) { skip("another order for this bet or market is already active"); continue; }
    try {
      const r = await k.rest(order, ins.id, expires);
      spent.kalshi = (spent.kalshi ?? 0) + stake + fee;          // a resting order reserves its cost
      const status = r.fill_count >= count ? "filled" : "resting";
      await db.updateOrder(ins.id, { status, order_id: r.order_id, fill_count: r.fill_count, fill_cost: r.fill_cost,
                                     fees: r.fees, response: r.raw });
      await db.updateFeed(row.key, { order_status: status, order_venue: "kalshi", order_id: r.order_id });
      await db.log("info", `${status === "filled" ? "FILLED" : "RESTING"}: ${text}`);
      res.actions.push(`${status}: ${text}`);
      counted.push({ ...order, status, fill_cost: r.fill_cost, created_at: now.toISOString() });
    } catch (e) {
      if (rejected(e)) {
        await db.updateOrder(ins.id, { status: "unfilled", reason: `rejected by Kalshi: ${msg(e)}` });
        await db.log("warn", `RESTING ORDER REJECTED by Kalshi (nothing placed): ${text}: ${msg(e)}`);
        skip(`resting order refused by Kalshi (${msg(e).slice(0, 120)})`);
        continue;
      }
      await db.updateOrder(ins.id, { status: "error", reason: msg(e) });
      await db.log("error", `RESTING ORDER FAILED (not retried): ${text}: ${msg(e)}`);
      res.actions.push(`error: ${text}: ${msg(e)}`);
    }
  }
}

/** Today's realized P&L on these orders, graded from final scores at the number each was bet. */
export async function pnlFromGames(db: Db, orders: OrderRow[], dayStartMs: number): Promise<number> {
  if (!db.games) throw new Error("no games table to grade from");
  const todays = orders.filter((o) => Date.parse(o.tip_utc) >= dayStartMs - 6 * 3600e3);
  if (!todays.length) return 0;
  const games = new Map((await db.games([...new Set(todays.map((o) => o.game_id))])).map((g) => [g.id, g]));
  let pnl = 0;
  for (const o of todays) {
    const g = games.get(o.game_id);
    if (!g || !g.final || g.home_score == null || g.away_score == null) continue;
    const res = gradeAt(o, g);
    const stake = o.fill_cost ?? 0, payout = o.fill_count ?? 0, fees = o.fees ?? 0;
    pnl += res === "W" ? payout - stake - fees : res === "L" ? -stake - fees : 0;
  }
  return pnl;
}

export function gradeAt(o: OrderRow, g: GameRow): "W" | "L" | "P" {
  const hs = g.home_score!, as_ = g.away_score!;
  let margin: number;
  if (o.bet === "total") margin = o.side === "UNDER" ? o.line - (hs + as_) : (hs + as_) - o.line;
  else margin = (o.side === g.home ? hs - as_ : as_ - hs) + o.line;
  return margin > 0 ? "W" : margin < 0 ? "L" : "P";
}

function best(r: FeedRow) {
  return Math.max(-1, ...(r.quotes || []).map((q) => q.ev ?? -1));
}
function orderSide(q: Quote) {
  if (q.venue === "kalshi") return String(q.order_side);
  return q.order_side ? String(q.order_side) : "long";
}
const round2 = (x: number) => Math.round(x * 100) / 100;
const round4 = (x: number) => Math.round(x * 10000) / 10000;
const cents = (x: number) => `${Math.round(x * 100)}¢`;
const money = (x: number) => `${x < 0 ? "-" : ""}$${Math.abs(x).toFixed(2)}`;
const pct = (x: number) => `${(x * 100).toFixed(1)}%`;
const msg = (e: unknown) => (e instanceof Error ? e.message : String(e)).slice(0, 400);

// ------------------------------------------------------------------ Kalshi
function b64ToBytes(b64: string): Uint8Array {
  return Uint8Array.from(atob(b64), (c) => c.charCodeAt(0));
}
function bytesToB64(b: ArrayBuffer): string {
  let s = "";
  for (const x of new Uint8Array(b)) s += String.fromCharCode(x);
  return btoa(s);
}
function derLen(n: number): number[] {
  if (n < 0x80) return [n];
  const out: number[] = [];
  while (n > 0) { out.unshift(n & 0xff); n >>= 8; }
  return [0x80 | out.length, ...out];
}
/** RSA "BEGIN RSA PRIVATE KEY" (PKCS#1) → PKCS#8, which Web Crypto can import. */
export function pkcs1ToPkcs8(pkcs1: Uint8Array): Uint8Array {
  const algId = [0x30, 0x0d, 0x06, 0x09, 0x2a, 0x86, 0x48, 0x86, 0xf7, 0x0d, 0x01, 0x01, 0x01, 0x05, 0x00];
  const octet = [0x04, ...derLen(pkcs1.length), ...pkcs1];
  const body = [0x02, 0x01, 0x00, ...algId, ...octet];
  return new Uint8Array([0x30, ...derLen(body.length), ...body]);
}

export type Signer = (message: string) => Promise<string>;

/** Ed25519 (Kalshi's default) or RSA-PSS SHA-256 with a 32-byte salt — detected from the key itself. */
export async function kalshiSigner(pem: string): Promise<Signer> {
  const text = pem.replace(/\\n/g, "\n").trim();
  const der = b64ToBytes(text.replace(/-----[^-]+-----/g, "").replace(/\s+/g, ""));
  const enc = new TextEncoder();
  if (text.includes("BEGIN RSA PRIVATE KEY")) return rsa(pkcs1ToPkcs8(der));
  try {
    const key = await crypto.subtle.importKey("pkcs8", der as BufferSource, { name: "Ed25519" }, false, ["sign"]);
    return async (m) => bytesToB64(await crypto.subtle.sign({ name: "Ed25519" }, key, enc.encode(m)));
  } catch {
    return rsa(der);
  }
  async function rsa(pkcs8: Uint8Array): Promise<Signer> {
    const key = await crypto.subtle.importKey("pkcs8", pkcs8 as BufferSource, { name: "RSA-PSS", hash: "SHA-256" }, false, ["sign"]);
    return async (m) => bytesToB64(await crypto.subtle.sign({ name: "RSA-PSS", saltLength: 32 }, key, enc.encode(m)));
  }
}

export class Kalshi implements Venue {
  name = "Kalshi";
  constructor(private base: string, private keyId: string | undefined, private signer: Signer | undefined,
              private dataBase = base, private http: typeof fetch = fetch) {}
  get hasKeys() { return !!(this.keyId && this.signer); }

  private async call(method: string, path: string, body?: unknown, auth = true, base = this.base) {
    const url = base + path;
    const headers: Record<string, string> = { "Content-Type": "application/json", Accept: "application/json" };
    if (auth) {
      if (!this.keyId || !this.signer) throw new Error("no Kalshi API keys in Supabase secrets");
      const ts = String(Date.now());
      headers["KALSHI-ACCESS-KEY"] = this.keyId;
      headers["KALSHI-ACCESS-TIMESTAMP"] = ts;
      headers["KALSHI-ACCESS-SIGNATURE"] = await this.signer(ts + method + new URL(url).pathname);
    }
    const r = await this.http(url, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const txt = await r.text();
    if (!r.ok) throw new Error(`Kalshi ${method} ${new URL(url).pathname} → ${r.status}: ${txt.slice(0, 300)}`);
    return txt ? JSON.parse(txt) : {};
  }

  async ask(q: Quote) {
    // dry runs read real (prod) prices; live reads the exchange the order goes to
    const js = await this.call("GET", `/markets/${q.ticker}`, undefined, false, this.dataBase);
    const m = js.market || {};
    const raw = q.order_side === "no" ? m.no_ask_dollars : m.yes_ask_dollars;
    const ask = raw == null ? null : Number(raw);
    return { ask: m.status && !["active", "open"].includes(m.status) ? null : ask, market: String(q.ticker), detail: m };
  }
  fee(p: number, c: number) { return kalshiFee(p, c); }

  /** Good-till-canceled order at the limit, expiring at `expiresUnix`; it waits in the book for the price to come to us. */
  async rest(o: OrderRow, clientId: string, expiresUnix: number) {
    const buyNo = o.order_side === "no";
    const yes = round4(buyNo ? 1 - o.limit_price : o.limit_price);
    const r = await this.call("POST", "/portfolio/events/orders", {
      ticker: o.market, side: buyNo ? "ask" : "bid", count: String(o.count), price: yes.toFixed(4),
      time_in_force: "good_till_canceled", expiration_time: Math.floor(expiresUnix),
      self_trade_prevention_type: "taker_at_cross", client_order_id: clientId,
    });
    if (!r.order_id) throw new Error(`Kalshi returned no order id: ${JSON.stringify(r).slice(0, 300)}`);
    const fill = Number(r.fill_count ?? 0);
    const avg = Number(r.average_fill_price ?? 0);
    const per = fill > 0 ? (buyNo ? (avg >= yes - 1e-6 ? 1 - avg : avg) : avg) : 0;
    return { order_id: String(r.order_id), fill_count: fill, fill_cost: round2(fill * per),
             fees: round2(fill * Number(r.average_fee_paid ?? 0)), raw: r };
  }
  async getOrder(orderId: string) {
    const js = await this.call("GET", `/portfolio/orders/${orderId}`);
    const o = js.order || {};
    const n = (x: unknown) => Number(x ?? 0) || 0;
    return { status: String(o.status ?? "unknown"), fill_count: n(o.fill_count_fp ?? o.fill_count),
             fill_cost: round2(n(o.taker_fill_cost_dollars) + n(o.maker_fill_cost_dollars)),
             fees: round2(n(o.taker_fees_dollars) + n(o.maker_fees_dollars)), raw: o };
  }
  async cancel(orderId: string) { await this.call("DELETE", `/portfolio/events/orders/${orderId}`); }
  async balance() {
    const js = await this.call("GET", "/portfolio/balance");
    return js.balance_dollars != null ? Number(js.balance_dollars) : Number(js.balance) / 100;
  }
  async pnlToday(markets: Set<string>, dayStartMs: number) {
    let cursor = "", pnl = 0;
    for (let page = 0; page < 20; page++) {
      const js = await this.call("GET",
        `/portfolio/settlements?limit=1000&min_ts=${Math.floor(dayStartMs / 1000)}${cursor ? "&cursor=" + cursor : ""}`);
      for (const st of js.settlements || []) {
        if (!markets.has(st.ticker)) continue;
        pnl += Number(st.revenue || 0) / 100 - Number(st.yes_total_cost_dollars || 0) -
          Number(st.no_total_cost_dollars || 0) - Number(st.fee_cost || 0);
      }
      cursor = js.cursor || "";
      if (!cursor) break;
    }
    return pnl;
  }
  async order(o: OrderRow, clientId: string) {
    // Kalshi V2 orders (the old /portfolio/orders address now answers 410). V2 quotes everything from the YES side:
    // buying YES at p = "bid" at p; buying NO at p = selling YES ("ask") at 1 - p.
    const buyNo = o.order_side === "no";
    const yesLimit = round4(buyNo ? 1 - o.limit_price : o.limit_price);
    const body = {
      ticker: o.market, side: buyNo ? "ask" : "bid", count: String(o.count), price: yesLimit.toFixed(4),
      time_in_force: "immediate_or_cancel", self_trade_prevention_type: "taker_at_cross", client_order_id: clientId,
    };
    const ord = await this.call("POST", "/portfolio/events/orders", body);
    if (!ord.order_id) throw new Error(`Kalshi returned no order id: ${JSON.stringify(ord).slice(0, 300)}`);
    const fill = Number(ord.fill_count ?? 0);
    let per = 0;
    if (fill > 0) {
      const avg = Number(ord.average_fill_price);
      if (!Number.isFinite(avg)) throw new Error(`Kalshi filled ${fill} but sent no average price: ${JSON.stringify(ord).slice(0, 300)}`);
      // the average is on the YES side; a sell of YES fills at or above our limit, so a NO buy costs 1 - avg.
      // (if it ever comes back on the NO side instead, it'll be at or below our NO limit — use it as is)
      per = buyNo ? (avg >= yesLimit - 1e-6 ? 1 - avg : avg) : avg;
    }
    const fees = fill * Number(ord.average_fee_paid ?? 0);
    return { order_id: String(ord.order_id), fill_count: fill, fill_cost: round2(fill * per), fees: round2(fees), raw: ord };
  }
}

// ------------------------------------------------------------------ BetOpenly (peer-to-peer offers)
// How Caden's Exchange Scanner bets there (since early Oct 2026): x-api-key header; open-bets lists every offer;
// re-check an offer's status right before accepting (no price lock); accept {amount}. ~1% of winnings as fee.
export const BETOPENLY_BASE = "https://public.betopenly.com";
const TEAM_NICKNAMES: Record<string, string> = {
  ATL: "Hawks", BOS: "Celtics", BKN: "Nets", CHA: "Hornets", CHI: "Bulls", CLE: "Cavaliers", DAL: "Mavericks",
  DEN: "Nuggets", DET: "Pistons", GSW: "Warriors", HOU: "Rockets", IND: "Pacers", LAC: "Clippers", LAL: "Lakers",
  MEM: "Grizzlies", MIA: "Heat", MIL: "Bucks", MIN: "Timberwolves", NOP: "Pelicans", NYK: "Knicks", OKC: "Thunder",
  ORL: "Magic", PHI: "76ers", PHX: "Suns", POR: "Trail Blazers", SAC: "Kings", SAS: "Spurs", TOR: "Raptors",
  UTA: "Jazz", WAS: "Wizards",
};
/** Our team code from whatever a feed calls the team (abbreviation or name). */
export function teamCode(t: { abbreviation?: string; displayName?: string; display_name?: string; name?: string }
                         | string | undefined | null): string | null {
  if (!t) return null;
  const abbr = typeof t === "string" ? t : (t.abbreviation ?? "");
  const name = typeof t === "string" ? t : (t.displayName ?? t.display_name ?? t.name ?? "");
  const a = polyCode(abbr || "");
  if (TEAM_NICKNAMES[a]) return a;
  const low = name.toLowerCase();
  for (const [code, nick] of Object.entries(TEAM_NICKNAMES)) if (low.includes(nick.toLowerCase())) return code;
  return null;
}
const SPREAD_WINDOW = 4, TOTAL_WINDOW = 6;
const toMs = (t: unknown) => { const n = Number(t); return n > 1e12 ? n : n * 1000; };

export class BetOpenly implements Venue {
  name = "BetOpenly";
  gradeFromGames = true;
  private book: any[] | null = null;
  private notes = new Map<string, string[]>();
  /** What BetOpenly had on this game and why each offer was passed over (for the log). */
  explain(row: FeedRow): string {
    const n = this.notes.get(row.key);
    if (!n) return "";
    return n.length ? `BetOpenly offers on this game: ${n.join("; ")}` : `BetOpenly: no offers on this game (${this.book?.length ?? 0} open offers in all)`;
  }
  constructor(private key?: string, private http: typeof fetch = fetch, private nowMs = () => Date.now()) {}
  get hasKeys() { return !!this.key; }

  private async call(method: string, path: string, body?: unknown) {
    if (!this.key) throw new Error("no BETOPENLY_KEY in Supabase secrets");
    const r = await this.http(BETOPENLY_BASE + path, {
      method, body: body === undefined ? undefined : JSON.stringify(body),
      headers: { "x-api-key": this.key, "content-type": "application/json", accept: "application/json" },
    });
    const txt = await r.text();
    if (r.status !== 200 && r.status !== 201) throw new Error(`BetOpenly ${method} ${path} → ${r.status}: ${txt.slice(0, 300)}`);
    return txt ? JSON.parse(txt) : {};
  }

  /** One live game offer in our terms, or null (parlays, futures, expired, not NBA full game...). */
  parse(b: any): { id: string; team: string | null; kind: string; line: number | null; odds: number; available: number;
                   eventMs: number; home: string | null; away: string | null } | null {
    if (b?.betType !== "GAME" || !Array.isArray(b.details) || b.details.length !== 1) return null;
    if (b.betStatus !== "PENDING") return null;
    if (b.expiryDate && toMs(b.expiryDate) <= this.nowMs()) return null;
    const d = b.details[0];
    if (String(d.league ?? "").toUpperCase() !== "NBA" || String(d.timeframe ?? "").toLowerCase() !== "game") return null;
    const kind0 = String(d.betType ?? "").toLowerCase();
    const kind = kind0 === "to win" ? "moneyline" : kind0;
    if (!["spread", "moneyline", "over", "under"].includes(kind)) return null;
    const eventMs = d.eventDateIso ? Date.parse(d.eventDateIso) : toMs(d.eventDate);
    return { id: String(b.id), team: teamCode(d.teamAbbv ?? d.team), kind,
             line: d.lines == null ? null : Number(d.lines), odds: Number(d.odds), available: Number(b.availableAmount ?? 0),
             eventMs, home: teamCode(d.homeTeam), away: teamCode(d.awayTeam) };
  }

  async offers(row: FeedRow): Promise<Quote[]> {
    const m = row.model;
    if (!m) return [];
    if (!this.book) {
      const js = await this.call("GET", "/v1/partners/bets/open-bets");
      this.book = Array.isArray(js) ? js : (js.data ?? []);
    }
    const out: Quote[] = [];
    const notes: string[] = [];
    this.notes.set(row.key, notes);
    for (const b of this.book!) {
      const d0 = Array.isArray(b?.details) ? b.details[0] : null;
      const teams = d0 ? [teamCode(d0.homeTeam), teamCode(d0.awayTeam)] : [];
      const onGame = teams.includes(row.home) && teams.includes(row.away);
      const desc = onGame ? `${d0.teamAbbv ?? d0.team ?? "?"} ${d0.betType ?? "?"} ${d0.lines ?? ""} ${d0.odds > 0 ? "+" : ""}${d0.odds} ($${b.availableAmount ?? "?"})` : "";
      const pass = (why: string) => { if (onGame && notes.length < 12) notes.push(`${desc}: ${why}`); };
      const o = this.parse(b);
      if (!o) { pass(`not a live full-game single bet (${b?.betType}/${b?.betStatus}/${d0?.timeframe}/${b?.details?.length} legs)`); continue; }
      if (o.home !== row.home || o.away !== row.away) continue;
      if (Math.abs(o.eventMs - Date.parse(row.tip_utc)) > 6 * 3600e3) { pass("different date"); continue; }
      if (!Number.isFinite(o.odds) || o.odds === 0 || o.available <= 0) { pass("no odds or nothing left"); continue; }
      let line: number, contract: string;
      if (m.bet === "total") {
        if (o.kind !== m.side.toLowerCase() || o.line == null) { pass("not our bet"); continue; }
        line = o.line; contract = `${m.side === "OVER" ? "Over" : "Under"} ${line}`;
      } else {
        if (o.team !== m.side) { pass("other side"); continue; }
        if (o.kind === "moneyline") { line = -0.5; contract = `${m.side} to win`; }
        else if (o.kind === "spread" && o.line != null) { line = o.line; contract = `${m.side} ${line > 0 ? "+" : ""}${line}`; }
        else { pass("not our bet"); continue; }
      }
      if (line < m.lo - 1e-9 || line > m.hi + 1e-9) { pass(`number outside the system (${m.lo} to ${m.hi})`); continue; }
      if (Math.abs(line - m.base) > (m.bet === "total" ? TOTAL_WINDOW : SPREAD_WINDOW)) { pass("too far from the model's number"); continue; }
      const cost = americanCost(o.odds);
      const { win, push } = outcomeAt(m, line);
      if (m.market != null) {                                                   // way off the market = bad data
        const mk = outcomeAt({ ...m, win0: 0.5, base: m.market }, line);
        if (cost < mk.win / (1 - mk.push) - 0.10) { pass("price too good to be real (bad data)"); continue; }
      }
      const all_in = cost + this.fee(cost, 1);
      const ev = win * (1 / all_in - 1) - (1 - win - push);
      pass(`expected return ${pct(ev)}`);
      out.push({ venue: "betopenly", venue_name: "BetOpenly", contract, side: m.side, line, american: o.odds,
                 cost: Math.round(cost * 10000) / 10000, win, ev: Math.round(ev * 10000) / 10000,
                 max_cost: win / (1 - push), size: o.available, offer_id: o.id, ticker: o.id, order_side: "accept",
                 ml: o.kind === "moneyline" });
    }
    return out.sort((a, b) => b.ev - a.ev);
  }

  async ask(q: Quote, row: FeedRow) {
    const js = await this.call("GET", `/v1/partners/bets/${q.offer_id}/status`);
    const b = js?.data ?? js;
    const d = (b?.details ?? [])[0] ?? {};
    const market = `bo:${row.away}@${row.home}:${row.bet}:${q.side}:${q.line}`;
    if (b?.betStatus !== "PENDING" || Number(d.odds) !== Number(q.american)) {
      return { ask: null, market, detail: { betStatus: b?.betStatus, odds: d.odds } };
    }
    return { ask: Math.round(americanCost(Number(d.odds)) * 10000) / 10000, market, detail: { available: b.availableAmount } };
  }
  /** ~1% of winnings. */
  fee(p: number, c: number) { return Math.round(0.01 * c * (1 - p) * 100) / 100; }
  async balance() {
    const js = await this.call("GET", "/v1/partners/funds");
    return Number(js.availableFunds ?? js.data?.availableFunds);
  }
  async pnlToday(): Promise<number> { throw new Error("BetOpenly P&L is graded from final scores"); }
  async order(o: OrderRow, _clientId: string, q?: Quote) {
    if (!q?.offer_id) throw new Error("no BetOpenly offer id");
    // re-check right before accepting: there's no price lock
    const st = await this.call("GET", `/v1/partners/bets/${q.offer_id}/status`);
    const b = st?.data ?? st;
    const d = (b?.details ?? [])[0] ?? {};
    if (b?.betStatus !== "PENDING") throw new Error(`offer is ${b?.betStatus}, not PENDING`);
    if (Number(d.odds) !== Number(q.american)) throw new Error(`odds changed ${q.american} → ${d.odds}`);
    const amount = Math.round(Math.min(o.stake, Number(b.availableAmount ?? o.stake)) * 100) / 100;
    if (amount <= 0) throw new Error("nothing left to take on this offer");
    const js = await this.call("POST", `/v1/partners/bets/${q.offer_id}/accept`, { amount });
    const r = js?.data ?? js;
    const accepted = Number(r.acceptedAmount ?? 0), toWin = Number(r.amountToWin ?? 0);
    if (!r.childBetId) throw new Error(`BetOpenly returned no bet id: ${JSON.stringify(js).slice(0, 300)}`);
    const fill = accepted >= o.stake - 0.01 ? o.count : Math.floor(o.count * accepted / o.stake);
    return { order_id: String(r.childBetId), fill_count: fill, fill_cost: accepted,
             fees: Math.round(0.01 * toWin * 100) / 100, raw: { offer: q.offer_id, ...r } };
  }
}


// ------------------------------------------------------------------ ProphetX (exchange, Trading API v4)
// Docs: docs.prophetx.co. Login with the access + secret key gives a short-lived token; at most 20 live
// sessions per key, so the token is cached in bet_state and refreshed instead of logging in every run.
// Prices are American odds; "value" = dollars you can stake at that price right now. Orders are
// buy-only, quantity = stake in dollars, fillOrKill fills what's there at your price or better.
export const PROPHETX_BASE = "https://cash.api.prophetx.co/partner";
export const PROPHETX_FEE = 0.02;            // reported 2% of net winnings (not in the official docs yet)

export class ProphetX implements Venue {
  name = "ProphetX";
  gradeFromGames = true;
  private token: string | null = null;
  private nba: { events: any[] } | null = null;
  private marketsCache = new Map<string, any[]>();
  notes = new Map<string, string[]>();
  constructor(private access?: string, private secret?: string, private db?: Db, private http: typeof fetch = fetch,
              private nowMs = () => Date.now(), private sleep = (ms: number) => new Promise((r) => setTimeout(r, ms))) {}
  get hasKeys() { return !!(this.access && this.secret); }

  private async raw(method: string, path: string, body?: unknown, auth = true) {
    const headers: Record<string, string> = { "content-type": "application/json", accept: "application/json" };
    if (auth) headers.authorization = `Bearer ${await this.auth()}`;
    const r = await this.http(PROPHETX_BASE + path, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const txt = await r.text();
    let js: any = {};
    try { js = txt ? JSON.parse(txt) : {}; } catch { js = { raw: txt }; }
    return { status: r.status, js };
  }
  private async call(method: string, path: string, body?: unknown) {
    const { status, js } = await this.raw(method, path, body);
    if (status < 200 || status >= 300) throw new Error(`ProphetX ${method} ${path} → ${status}: ${JSON.stringify(js).slice(0, 300)}`);
    return js?.data ?? js;
  }

  /** A valid access token: cached → refreshed → fresh login (in that order). */
  private async auth(): Promise<string> {
    if (this.token) return this.token;
    if (!this.hasKeys) throw new Error("no PROPHETX_ACCESS_KEY / PROPHETX_SECRET_KEY in Supabase secrets");
    const now = this.nowMs();
    const ms = (t: unknown, fallback: number) => { const n = Number(t); return !n ? fallback : n > 1e12 ? n : n * 1000; };
    const st = (await this.db?.getState?.("prophetx_token")) as any;
    if (st?.access_token && now < (st.access_until ?? 0) - 60e3) return (this.token = st.access_token);
    if (st?.refresh_token && now < (st.refresh_until ?? 0) - 60e3) {
      const r = await this.raw("POST", "/auth/refresh", { refresh_token: st.refresh_token }, false);
      const d = r.js?.data;
      if (r.status === 200 && d?.access_token) {
        await this.db?.setState?.("prophetx_token", { ...st, access_token: d.access_token,
          access_until: ms(d.access_expire_time, now + 8 * 60e3) });
        return (this.token = d.access_token);
      }
    }
    const r = await this.raw("POST", "/auth/login", { access_key: this.access, secret_key: this.secret }, false);
    const d = r.js?.data;
    if (r.status !== 200 || !d?.access_token) throw new Error(`ProphetX login → ${r.status}: ${JSON.stringify(r.js).slice(0, 200)}`);
    await this.db?.setState?.("prophetx_token", { access_token: d.access_token, refresh_token: d.refresh_token,
      access_until: ms(d.access_expire_time, now + 8 * 60e3), refresh_until: ms(d.refresh_expire_time, now + 2 * 86400e3) });
    return (this.token = d.access_token);
  }

  /** This game's ProphetX event (matched by both teams), or null. */
  private async event(row: FeedRow) {
    if (!this.nba) {
      const t = await this.call("GET", "/mm/get_tournaments?has_active_events=true");
      const nba = (t?.tournaments ?? []).find((x: any) => /^nba$/i.test(String(x.name).trim()))
        ?? (t?.tournaments ?? []).find((x: any) => /\bnba\b/i.test(x.name) && !/g league|summer|wnba|cup/i.test(x.name));
      if (!nba) throw new Error("no NBA tournament on ProphetX right now");
      const ev = await this.call("GET", `/mm/get_sport_events?tournament_id=${nba.id}`);
      this.nba = { events: ev?.sport_events ?? [] };
    }
    for (const e of this.nba.events) {
      const side = (k: string) => (e.competitors ?? []).find((c: any) => c.side === k);
      const h = teamCode(side("home")), a = teamCode(side("away"));
      if (h === row.home && a === row.away && Math.abs(Date.parse(e.scheduled) - Date.parse(row.tip_utc)) < 6 * 3600e3) {
        return e;
      }
    }
    return null;
  }
  private async markets(eventId: string, fresh = false) {
    if (fresh || !this.marketsCache.has(eventId)) {
      const d = await this.call("GET", `/v4/mm/get_markets?event_id=${eventId}`);
      this.marketsCache.set(eventId, d?.markets ?? []);
    }
    return this.marketsCache.get(eventId)!;
  }

  /** Every selection on this game as {team or OVER/UNDER, kind, line, price, value, strike_id}. */
  selections(markets: any[], ev: any) {
    const comp = new Map<number, string | null>((ev.competitors ?? []).map((c: any) => [c.id, teamCode(c)]));
    const out: { who: string | null; kind: string; line: number | null; price: number; value: number; strike_id: string }[] = [];
    const walk = (m: any) => {
      for (const group of m.selections ?? []) {
        for (const sel of (Array.isArray(group) ? group : [group])) {
          if (!sel?.strike_id || !Number.isFinite(Number(sel.price))) continue;
          const name = `${sel.name ?? ""} ${sel.display_name ?? ""}`.toLowerCase();
          const who = m.type === "total" ? (/\bover\b/.test(name) ? "OVER" : /\bunder\b/.test(name) ? "UNDER" : null)
            : comp.get(sel.competitor_id) ?? teamCode(sel.name);
          out.push({ who, kind: m.type, line: sel.strike == null || m.type === "moneyline" ? null : Number(sel.strike),
                     price: Number(sel.price), value: Number(sel.value ?? 0), strike_id: String(sel.strike_id) });
        }
      }
      for (const sub of m.market_strikes ?? []) if (sub && typeof sub === "object") walk({ ...sub, type: sub.type ?? m.type });
    };
    for (const m of markets) if (["moneyline", "spread", "total"].includes(m.type) && (!m.status || /active|open/i.test(m.status))) walk(m);
    return out;
  }

  explain(row: FeedRow): string {
    const n = this.notes.get(row.key);
    return !n ? "" : n.length ? `ProphetX on this game: ${n.join("; ")}` : "ProphetX: nothing on this game";
  }

  async offers(row: FeedRow): Promise<Quote[]> {
    const m = row.model;
    if (!m) return [];
    const notes: string[] = [];
    this.notes.set(row.key, notes);
    const ev = await this.event(row);
    if (!ev) { notes.push("game not listed"); return []; }
    const out: Quote[] = [];
    for (const x of this.selections(await this.markets(String(ev.event_id)), ev)) {
      const desc = `${x.who ?? "?"} ${x.kind} ${x.line ?? ""} ${x.price > 0 ? "+" : ""}${x.price} ($${x.value})`;
      const pass = (why: string) => { if (notes.length < 12) notes.push(`${desc}: ${why}`); };
      if (m.bet === "total" ? x.kind !== "total" || x.who !== m.side : x.kind === "total" || x.who !== m.side) continue;
      let line: number;
      if (x.kind === "moneyline") line = -0.5;
      else if (x.line == null) { pass("no number"); continue; }
      else line = x.line;
      if (x.value < 1) { pass("nothing to take"); continue; }
      // ProphetX's docs don't say which way a spread number is signed for each side, so check it against
      // the market: if the market has us getting points and ProphetX says we're laying them (or the
      // reverse), the number is read backwards — skip it rather than bet the wrong line.
      if (x.kind === "spread" && m.market != null && Math.abs(m.market) >= 1 && Math.sign(line) !== Math.sign(m.market)) {
        pass(`number looks flipped vs the market (${m.market > 0 ? "+" : ""}${m.market})`); continue;
      }
      if (line < m.lo - 1e-9 || line > m.hi + 1e-9) { pass(`number outside the system (${m.lo} to ${m.hi})`); continue; }
      if (Math.abs(line - m.base) > (m.bet === "total" ? TOTAL_WINDOW : SPREAD_WINDOW)) { pass("too far from the model's number"); continue; }
      const cost = americanCost(x.price);
      const { win, push } = outcomeAt(m, line);
      if (m.market != null) {
        const mk = outcomeAt({ ...m, win0: 0.5, base: m.market }, line);
        if (cost < mk.win / (1 - mk.push) - 0.10) { pass("price too good to be real (bad data)"); continue; }
      }
      const ev_ = win * (1 / (cost + this.fee(cost, 1)) - 1) - (1 - win - push);
      pass(`expected return ${pct(ev_)}`);
      const contract = m.bet === "total" ? `${m.side === "OVER" ? "Over" : "Under"} ${line}`
        : x.kind === "moneyline" ? `${m.side} to win` : `${m.side} ${line > 0 ? "+" : ""}${line}`;
      out.push({ venue: "prophetx", venue_name: "ProphetX", contract, side: m.side, line, american: x.price,
                 cost: round4(cost), win, ev: round4(ev_), max_cost: win / (1 - push), size: x.value,
                 offer_id: x.strike_id, ticker: `${ev.event_id}:${x.strike_id}`, order_side: "buy", ml: x.kind === "moneyline" });
    }
    return out.sort((a, b) => b.ev - a.ev);
  }

  async ask(q: Quote, row: FeedRow) {
    const market = `px:${row.away}@${row.home}:${row.bet}:${q.side}:${q.line}`;
    const ev = await this.event(row);
    if (!ev) return { ask: null, market };
    const x = this.selections(await this.markets(String(ev.event_id), true), ev).find((s) => s.strike_id === q.offer_id);
    if (!x || x.value < 1) return { ask: null, market, detail: { gone: true } };
    return { ask: round4(americanCost(x.price)), market, detail: { price: x.price, value: x.value } };
  }
  /** ~2% of winnings (c = payout dollars, p = price per $1 payout). */
  fee(p: number, c: number) { return Math.round(PROPHETX_FEE * c * (1 - p) * 100) / 100; }
  async balance() { return Number((await this.call("GET", "/v4/mm/get_balance"))?.balance); }
  async pnlToday(): Promise<number> { throw new Error("ProphetX P&L is graded from final scores"); }

  async order(o: OrderRow, clientId: string, q?: Quote) {
    if (!q?.offer_id || q.american == null) throw new Error("no ProphetX selection");
    const quantity = Math.round(o.stake * 100) / 100;
    const sub = await this.call("POST", "/v4/mm/submit_order", { external_id: clientId, strike_id: q.offer_id,
      price: q.american, quantity, order_strategy: "fillOrKill" });
    const id = sub?.order?.order_id;
    if (!id) throw new Error(`ProphetX returned no order id: ${JSON.stringify(sub).slice(0, 300)}`);
    // the reply comes back before matching; poll until nothing is left open (about 2-7 s for fillOrKill)
    let ord: any = null;
    for (let i = 0; i < 10; i++) {
      await this.sleep(i < 3 ? 1500 : 2500);
      const r = await this.raw("GET", `/v4/mm/get_order/${id}`);
      if (r.status === 404) continue;
      if (r.status !== 200) throw new Error(`ProphetX get_order → ${r.status}`);
      ord = r.js?.data ?? r.js;
      if (Number(ord?.open_quantity ?? 1) === 0 || /cancel|closed|settled/i.test(String(ord?.status))) break;
    }
    if (!ord) throw new Error(`ProphetX order ${id}: status not visible after 20 s (check it on ProphetX)`);
    const filled = Number(ord.filled_quantity ?? (ord.matching_status === "fully_matched" ? ord.quantity : 0)) || 0;
    let profit = 0;
    if (filled > 0) {
      const md = await this.call("GET", `/v4/mm/get_order_matched_detail?order_id=${id}`);
      profit = (md?.matching_details ?? []).reduce((t: number, m: any) => t + Number(m.profit ?? 0), 0);
      if (!profit) profit = filled * (1 / americanCost(q.american) - 1);
    }
    const payout = filled + profit;
    return { order_id: String(id), fill_count: filled > 0 ? Math.round(payout * 100) / 100 : 0, fill_cost: filled,
             fees: Math.round(PROPHETX_FEE * profit * 100) / 100, raw: { order: ord, profit } };
  }
}

// ------------------------------------------------------------------ Supabase wiring
// deno-lint-ignore no-explicit-any
export function supabaseDb(sb: any): Db {
  const must = (r: { error: unknown; data?: unknown }) => {
    if (r.error) throw new Error(JSON.stringify(r.error));
    return r.data;
  };
  return {
    async settings() { return must(await sb.from("bet_settings").select("*").eq("id", 1).single()) as Settings; },
    async updateSettings(p) { must(await sb.from("bet_settings").update({ ...p, updated_at: new Date().toISOString() }).eq("id", 1)); },
    async feed(minUpdated, minTip) {
      return must(await sb.from("bet_feed").select("*").in("status", ["bet", "limit"]).gte("updated_at", minUpdated)
        .gte("tip_utc", minTip)) as FeedRow[];
    },
    async orders(since) { return must(await sb.from("bet_orders").select("*").gte("created_at", since)) as OrderRow[]; },
    async insertOrder(row) {
      const r = await sb.from("bet_orders").insert(row).select("id").single();
      if (r.error?.code === "23505") return { conflict: true };
      return { id: (must(r) as { id: string }).id };
    },
    async updateOrder(id, p) { must(await sb.from("bet_orders").update({ ...p, updated_at: new Date().toISOString() }).eq("id", id)); },
    async updateFeed(key, p) { must(await sb.from("bet_feed").update(p).eq("key", key)); },
    async log(level, message, data) { await sb.from("bet_log").insert({ level, message, data: data ?? null }); },
    async games(ids) {
      return must(await sb.from("games").select("id,home,away,home_score,away_score,final,tip_utc").in("id", ids)) as GameRow[];
    },
    async getState(key) {
      const r = must(await sb.from("bet_state").select("value").eq("key", key).maybeSingle()) as { value: unknown } | null;
      return r ? r.value : null;
    },
    async setState(key, value) {
      must(await sb.from("bet_state").upsert({ key, value, updated_at: new Date().toISOString() }));
    },
  };
}

/** Read-only check of every key in the secrets (balances only). Never places anything. */
export function keyCheck(db?: Db): () => Promise<string> {
  return async () => {
    const out: string[] = [];
    for (const env of ["prod", "demo"] as const) {
      const id = Deno.env.get(env === "prod" ? "KALSHI_KEY_ID" : "KALSHI_DEMO_KEY_ID");
      const pem = Deno.env.get(env === "prod" ? "KALSHI_PRIVATE_KEY" : "KALSHI_DEMO_PRIVATE_KEY");
      if (!id || !pem) { out.push(`Kalshi ${env}: no keys`); continue; }
      try {
        const k = new Kalshi(KALSHI_BASE[env], id, await kalshiSigner(pem));
        out.push(`Kalshi ${env}: keys work, balance ${money(await k.balance())}`);
      } catch (e) { out.push(`Kalshi ${env}: ${msg(e)}`); }
    }
    if (Deno.env.get("BETOPENLY_KEY")) {
      try { out.push(`BetOpenly: key works, balance ${money(await new BetOpenly(Deno.env.get("BETOPENLY_KEY")).balance())}`); }
      catch (e) { out.push(`BetOpenly: ${msg(e)}`); }
    } else out.push("BetOpenly: no key");
    if (Deno.env.get("PROPHETX_ACCESS_KEY") && Deno.env.get("PROPHETX_SECRET_KEY")) {
      try {
        const px = new ProphetX(Deno.env.get("PROPHETX_ACCESS_KEY"), Deno.env.get("PROPHETX_SECRET_KEY"), db);
        out.push(`ProphetX: keys work, balance ${money(await px.balance())}`);
      } catch (e) { out.push(`ProphetX: ${msg(e)}`); }
    } else out.push("ProphetX: no keys");
    out.push(`POLYMARKET_KEY_ID: ${Deno.env.get("POLYMARKET_KEY_ID") ? "saved (dry run only)" : "not set"}`);
    return "Key check — " + out.join(" · ");
  };
}

export async function buildVenues(s: Settings, db?: Db): Promise<Record<string, Venue>> {
  const env = s.kalshi_env === "prod" ? "prod" : "demo";
  const keyId = Deno.env.get(env === "prod" ? "KALSHI_KEY_ID" : "KALSHI_DEMO_KEY_ID");
  const pem = Deno.env.get(env === "prod" ? "KALSHI_PRIVATE_KEY" : "KALSHI_DEMO_PRIVATE_KEY");
  const signer = keyId && pem ? await kalshiSigner(pem) : undefined;
  const dataBase = s.live_betting_on ? KALSHI_BASE[env] : KALSHI_BASE.prod;
  return {
    kalshi: new Kalshi(KALSHI_BASE[env], keyId, signer, dataBase),
    polymarket_us: new PolymarketUS(Deno.env.get("POLYMARKET_KEY_ID"), Deno.env.get("POLYMARKET_SECRET_KEY")),
    betopenly: new BetOpenly(Deno.env.get("BETOPENLY_KEY")),
    prophetx: new ProphetX(Deno.env.get("PROPHETX_ACCESS_KEY"), Deno.env.get("PROPHETX_SECRET_KEY"), db),
  };
}


Deno.serve(async (req) => {
  const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
    { auth: { persistSession: false } });
  const db = supabaseDb(sb);
  try {
    const s = await db.settings();
    if (req.headers.get("x-run-token") !== String(s.run_token)) return new Response("forbidden", { status: 403 });
    const out = await runOnce(db, await buildVenues(s, db), new Date(), keyCheck(db));
    return new Response(JSON.stringify(out), { headers: { "Content-Type": "application/json" } });
  } catch (e) {
    const m = (e instanceof Error ? e.message : String(e)).slice(0, 400);
    await db.log("error", `run crashed: ${m}`).catch(() => {});
    return new Response(JSON.stringify({ error: m }), { status: 500 });
  }
});
